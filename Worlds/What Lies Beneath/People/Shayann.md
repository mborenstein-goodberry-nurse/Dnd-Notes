## Description

**[[Shayann]]** is **[[Kiara Bashara]]'s** beloved brother, originally from **[[Riverrun]]** in the **[[Fey Forests]]**. His appearance, age, present location, and current circumstances have not been established.

When Kiara was approximately 20–30 years old, a priest of Xurxus accosted her with predatory intentions. Shayann killed the priest to protect his sister. The priest may have been named **Wren**, but this is unconfirmed.

Killing a priest was treated as an unforgivable offense by their community. Shayann fled and was banished; people were forbidden even to speak his name. Kiara has not seen him since. **She believes he is alive and that his actions were justified**, but his actual present condition is unknown.

// Normalize arrays, Dataview DataArrays, and single values.
function toArray(value) {
    if (value == null) return [];
    if (Array.isArray(value)) return value;
    if (typeof value.array === "function") return value.array();
    return [value];
}

// Resolve a wikilink or Dataview link to its actual file.
function resolveFile(value, sourcePath) {
    if (!value) return null;

    const raw = value.path
        ? value.path
        : String(value)
            .replace(/^!?\[\[|\]\]$/g, "")
            .split("|")[0];

    return app.metadataCache.getFirstLinkpathDest(
        raw,
        sourcePath
    );
}

const results = [];

for (const quest of quests) {
    const roles = [];

    // Check whether this NPC is the quest giver.
    const giver = resolveFile(
        quest.questGiver,
        quest.file.path
    );

    if (giver?.path === current.file.path) {
        roles.push("Quest Giver");
    }

    // Check general quest involvement.
    const involved = toArray(quest.people).some(value => {
        const file = resolveFile(value, quest.file.path);
        return file?.path === current.file.path;

    if (involved && !roles.includes("Quest Giver")) {
        roles.push("Involved");

    if (roles.length) {
        results.push({
            quest: quest.file.link,
            campaign: quest.campaign ?? "-",
            involvement: roles.join(", "),
            status: quest.status ?? "Unknown"
        });

}
